async function sendMessage() { const input = document.getElementById('user-input'); const text = input.value.trim(); if (!text) return; input.value = ""; const chatBox = document.getElementById('chat-box'); chatBox.innerHTML += `
${escapeHtml(text)}
`; history.push({ role: "user", content: text }); const replyId = "reply-" + Date.now(); chatBox.innerHTML += `
Scrittura in corso...
`; chatBox.scrollTop = chatBox.scrollHeight; try { isSpeaking = true; // Chiamata diretta all'endpoint pubblico per evitare errori 405 sul server const response = await fetch("https://text.pollinations.ai/", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ messages: [ { role: "system", content: `Ti chiami ${cfg.name}. Rispondi sempre in italiano in modo chiaro, amichevole e breve.` }, ...history ], model: cfg.model }) }); if (!response.ok) throw new Error("Errore di connessione"); const reply = await response.text(); history.push({ role: "assistant", content: reply }); document.getElementById(replyId).innerHTML = `${cfg.name}: ${escapeHtml(reply)}`; speakText(reply); } catch (err) { document.getElementById(replyId).innerHTML = `Errore di ricezione. Riprova.`; } finally { isSpeaking = false; chatBox.scrollTop = chatBox.scrollHeight; } }
